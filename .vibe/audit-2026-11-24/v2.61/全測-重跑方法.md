# 重跑方法（在 fulltest 資料夾用 Git Bash 執行）

需要：同層有 `snap-new`（要測的新版）和 `snap-v260`（716277f）。新版換掉時先重做快照，再把下面 `N` 指過去：
`mkdir ../snap-new2 && git -C "E:/捏捏史萊姆/nienie-slime" archive <提交> | tar -x -C ../snap-new2`
（eq.py、eq_far.py 的第一個參數是新版 web 夾；con.py、stx.py、mix.sh 直接帶夾子。）

```bash
cd <fulltest>; export PYTHONIOENCODING=utf-8
S=$(cd ..; pwd -W); N=$S/snap-new/prototype/web; O=$S/snap-v260/prototype/web
python mk.py && python mk2.py && python mk4.py        # 產生 eq.py、eq4.js、eq_far.py、eqh.js（mk3.py 只跑過一次，con.py 已改好，不要再跑）

# 1 整份自測 3 次（並行 3 個 Chrome，約 21 分鐘；每份約 1245 秒）
for i in 1 2 3; do python stx.py $N $((8994+i)) out/st_new$i.json > out/st_new$i.log 2>&1 & done; wait; cat out/st_new?.log
python cmpall.py out/st_new1.json out/st_new2.json out/st_new3.json | grep -v " 0 / "     # 三次哪裡不同
python stx.py $O 8995 out/st_v260.json                # v2.60 對照，約 12 分鐘

# 2 混合 250＋最糟 40（4 個 Chrome，約 11 分鐘）。預設＝換皮＋第二、三包，全開＝再加拉的三顆
bash mix.sh $N ; bash mix.sh $N _2                    # 第二個參數是字尾，重複跑量雜訊
python sum.py mix_def mix_all gone_def gone_all; python pair.py mix_def mix_def_2
bash mix_clean.sh $N _2                               # 建議用這個跟 W1 比：清掉開頁「先看你」的殘留，多一組「第二、三包關掉」（約 20 分鐘）
python pair.py cl_w1 cl_off23_2 cl_def_2; python pair.py cl_w1all cl_all_2      # cl_w1、cl_w1all 是 W1 那版同條件的結果（W1 原碼在舊暫存夾 w1/web61，8dd1c17 可以重做）
# --procs 不要改：每個 Chrome 輪到哪幾組會跟著變，逐組對照就對不起來

# 3 T70（單獨，約 9 分鐘一個，可並行）。預設、全開都在同一份輸出的 def／all；整份自測裡也有
python stx.py $N 8995 out/t70_new.json t70
python stx.py $O 8996 out/t70_v260.json t70 stf=$N/selftest.js "pre=window.GREET=window.GREET||{}"     # 要紅
python stx.py $N 8997 out/t70_mut.json t70 "mut=DEF = { ...OLD, skinU: 1 }=>DEF = { ...OLD, skinU: 0 }"  # 新版改回舊皮，要紅

# 4 開關全 0 對 v2.60（埠 8998、8999，各約 6、12、12 分鐘）
python eq.py $N      # 五隻×2 種子，每組 93 次狀態＋畫面
python eq_far.py $N  # 加拉到螢幕邊，有補點，每組 298 次
python remchk.py     # 證明 eq_far 真的有補點
python drv.py $O 8998 eq4.js out/eq4_v260a.json; python drv.py $N 8999 eq4.js out/eq4_new.json   # 四隻×2 種子，睡、走、聽節奏（120 段狀態、最後畫面、出聲）
python eqcmp.py out/eq4_v260a.json out/eq4_new.json

# 5 主控台（約 2 分鐘，會真的點畫面；結尾自己塞錯誤和 NaN 證明抓得到）
python con.py $N 8999 out/con_new.json
```
